@restmodel/core API ​

由公开入口自动生成的符号索引。函数参数、默认值和行为边界见客户端参考与操作声明。类型不替代运行时响应校验。

resource ​

创建资源能力定义，不执行 HTTP 请求。

ts
resource: { <const TConfig extends ResourceConfig>(config: TConfig): ResourceDefinition<unknown, TConfig>; <TModel = unknown>(): ResourceDefinition<TModel, { operations: readonly []; }>; <TModel = unknown>(config: ResourceConfig & { kind: "singleton"; }): ResourceDefinition<TModel, ResourceConfig & { kind: "singleton"; }>; <TModel = unknown>(config: ResourceConfig & { kind?: "collection" | undefined; }): ResourceDefinition<TModel, ResourceConfig & { kind: "collection"; }>; <TModel = unknown, const TConfig extends ResourceConfig = ResourceConfig>(config: TConfig): ResourceDefinition<TModel, TConfig>; }

defineResourcePreset ​

快照并冻结可复用预设，不绑定资源形态、不执行请求。

ts
defineResourcePreset: <const TConfig extends ResourcePreset>(config: TConfig) => ResourcePresetDefinition<TConfig>

ResourcePreset ​

可复用的能力声明，不提前绑定资源形态。

ts
interface ResourcePreset {
  /** 本层能力补丁；缺省或 undefined 沿用父预设，不接受 null。 */
  readonly operations?: Operations | undefined;
  /** 单一父预设；缺省或 undefined 无继承，不接受 null 或循环。 */
  readonly extends?: ResourcePreset | undefined;
}

ResourcePresetDefinition ​

预设快照保留原声明类型，以便推导继承能力。

ts
type ResourcePresetDefinition<TConfig extends ResourcePreset> = ResourcePreset & {
  readonly [presetType]: TConfig;
};

ResourceConfig ​

资源自身的声明。

ts
interface ResourceConfig extends ResourcePreset {
  /** 资源形态；缺省或 undefined 为 collection，不接受 null。 */
  readonly kind?: ResourceKind | undefined;
  /** 相对资源路径；默认使用声明键，不允许绝对路径。 */
  readonly path?: string;
  /** 服务规则局部覆盖。 */
  readonly scheme?: Scheme;
  /** 当前集合或单体的子资源。 */
  readonly children?: ResourceTree;
  /** 集合成员的子资源；单体不支持。 */
  readonly members?: ResourceTree;
  /** 当前路径上的特殊动作。 */
  readonly actions?: ActionTree;
  /** 集合成员动作；单体不支持。 */
  readonly memberActions?: ActionTree;
}

ResourceDefinition ​

冻结的声明结果，不是可发送请求的客户端。

ts
interface ResourceDefinition<
  TModel = unknown,
  TConfig extends ResourceConfig = ResourceConfig,
> {
  /** 已确定的资源形态。 */
  readonly kind: ResourceKind;
  /** 已归一化的只读能力映射。 */
  readonly operations: ResolvedOperations;
  /** 仅携带模型类型，不产生运行时字段。 */
  readonly [modelType]?: TModel;
  /** 仅携带原配置类型，不产生运行时字段。 */
  readonly [configType]?: TConfig;
  /** 已快照的资源局部配置，供客户端装配。 */
  readonly config?: Readonly<Omit<ResourceConfig, 'extends' | 'operations' | 'kind'>>;
}

standardOperations ​

标准操作默认方法及目标位置的唯一目录；同时驱动类型和星号展开。

ts
standardOperations: Readonly<{ readonly list: Readonly<{ method: "GET"; target: "collection"; }>; readonly get: Readonly<{ method: "GET"; target: "member"; }>; readonly post: Readonly<{ method: "POST"; target: "collection"; }>; readonly put: Readonly<{ method: "PUT"; target: "member"; }>; readonly patch: Readonly<{ method: "PATCH"; target: "member"; }>; readonly delete: Readonly<{ method: "DELETE"; target: "member"; }>; }>

ResourceConfigurationError ​

声明配置无效时抛出的稳定错误分类。

ts
ResourceConfigurationError: typeof ResourceConfigurationError

StandardOperation ​

标准操作名，直接来自唯一操作目录。

ts
type StandardOperation = keyof typeof standardOperations;

HttpMethod ​

支持覆盖的 HTTP 方法；必须大写。

ts
type HttpMethod = (typeof httpMethods)[number];

ResourceKind ​

集合（含成员）或单体资源。

ts
type ResourceKind = 'collection' | 'singleton';

Operations ​

能力声明的三种等价入口，数组采用追加语义。

ts
type Operations = '*' | readonly StandardOperation[] | OperationMap;

OperationMap ​

逐项补丁；false 关闭，星号仅接受 true。

ts
type OperationMap = Readonly<
  Partial<Record<StandardOperation, boolean | OperationOverride>> & { '*'?: true }
>;

OperationOverride ​

单项操作覆盖。

ts
interface OperationOverride {
  /** HTTP 方法；缺省保留继承值，否则使用目录默认值。 */
  readonly method?: HttpMethod;
  /** 仅适配当前已开放操作；不能授予能力。 */
  readonly scheme?: Scheme;
}

ResolvedOperation ​

单项能力的归一化结果。

ts
interface ResolvedOperation {
  /** 最终 HTTP 方法。 */
  readonly method: HttpMethod;
  /** 请求作用于集合、成员或单体。 */
  readonly target: 'collection' | 'member' | 'singleton';
  /** 当前操作的局部规则。 */
  readonly scheme?: Scheme;
}

ResolvedOperations ​

未出现的标准操作表示未开放。

ts
type ResolvedOperations = Readonly<Partial<Record<StandardOperation, ResolvedOperation>>>;

createClient ​

将资源声明装配为服务客户端，不执行请求。

ts
createClient: <const TResources extends ResourceTree, const TActions extends ActionTree = Readonly<Record<string, import("/Users/ivan/Codes/RESTModel/packages/core/src/action").ActionDefinition<unknown, unknown>>>>(config: ClientConfig<TResources, TActions>) => Client<TResources, TActions>

Client ​

客户端公共入口及控制面。

ts
type Client<TResources extends ResourceTree, TActions extends ActionTree = ActionTree> = {
  readonly [TKey in keyof TResources]: DefinitionReference<TResources[TKey]>;
} & {
  readonly $: {
    /** @param name 已声明资源名。 @returns 动态入口，不猜模型类型。 */
    resource(name: string): ResourceReference;
    /** 服务根特殊接口。 */
    readonly actions: ActionFunctions<TActions>;
    /** @param request 服务内 raw HTTP 配置。 @returns 原始响应，不套信封。 */
    request(request: {
      readonly method: HttpMethod;
      readonly path: string;
      readonly query?: Query;
      readonly body?: unknown;
      readonly options?: RequestOptions;
    }): Promise<HttpResponse>;
  };
};

ClientConfig ​

一个服务的设置，Transport 必须显式传入。

ts
interface ClientConfig<
  TResources extends ResourceTree = ResourceTree,
  TActions extends ActionTree = ActionTree,
> extends HttpOptions {
  /** 绝对 HTTP(S) 地址或浏览器服务根路径。 */
  readonly baseURL: string;
  /** 命名资源定义；不自动猜测任意属性。 */
  readonly resources: TResources;
  /** 服务规则；默认裸 body、id 身份、page/size。 */
  readonly scheme?: Scheme;
  /** 客户端默认请求设置。 */
  readonly defaults?: Omit<RequestOptions, 'result'>;
  /** 服务根级特殊接口。 */
  readonly actions?: TActions;
}

defineScheme ​

校验并快照服务规则，数组或未知字段被拒绝。

ts
defineScheme: (input: Scheme) => Scheme

Scheme ​

可继承的服务规则；只允许这些已知域，不做任意深合并。

ts
interface Scheme {
  /** 对象身份字段，默认 id；false 关闭推导。 */
  readonly identity?: string | false;
  /** 信封数据字段路径；false 使用整个 body。 */
  readonly envelope?: string | false;
  /** 显式业务成功判定；false 关闭继承判定。 */
  readonly success?: ((body: unknown) => boolean) | false;
  /** 响应转换；false 关闭，不允许异步回调伪装同步。 */
  readonly map?: ((data: unknown) => unknown) | false;
  /** 分页；false 关闭；缺省 page/size，仅显式 page 调用附加参数。 */
  readonly pagination?: PaginationScheme | false;
}

PaginationScheme ​

显式分页请求和响应字段；模式切换整段替换。

ts
interface PaginationScheme {
  /** 分页模式。 */
  readonly mode: 'page' | 'offset' | 'cursor';
  /** 页码、偏移或 cursor 的查询字段，默认分别 page/offset/cursor。 */
  readonly parameter?: string;
  /** 大小字段；默认 page 为 size，其他为 limit。 */
  readonly size?: string;
  /** 页码基数，默认 1，仅 page 适用。 */
  readonly base?: 0 | 1;
  /** 列表字段路径，缺省要求数据本身为数组。 */
  readonly items?: string;
  /** 总数字段路径；缺省未知。 */
  readonly total?: string;
  /** 下一 cursor 字段路径。 */
  readonly next?: string;
}

action ​

校验并冻结特殊接口声明。

ts
action: <TInput = unknown, TOutput = unknown>(config: ActionConfig) => ActionDefinition<TInput, TOutput>

ActionConfig ​

特殊接口输入配置。

ts
interface ActionConfig {
  /** 必须显式给出 HTTP 方法。 */
  readonly method: HttpMethod;
  /** 相对所属路径；/ 开头表示服务路径根，禁止跨 origin。 */
  readonly path?: string;
  /** 默认 GET/DELETE/HEAD/OPTIONS 用 query，其他用 body。 */
  readonly input?: 'query' | 'body';
  /** 单项服务适配覆盖。 */
  readonly scheme?: Scheme;
}

ActionDefinition ​

不可变动作声明；泛型仅携带类型，不校验响应。

ts
interface ActionDefinition<TInput = unknown, TOutput = unknown> extends ActionConfig {
  /** 编译期输入输出标记。 */
  readonly [actionTypes]?: { input: TInput; output: TOutput };
}

ActionFunctions ​

按动作声明推导输入与输出；默认 unknown 保持动态入口。

ts
type ActionFunctions<
  TActions extends Readonly<Record<string, ActionDefinition>>,
  TResponse extends boolean = false,
> = {
  readonly [TKey in keyof TActions]: TActions[TKey] extends ActionDefinition<
    infer TInput,
    infer TOutput
  >
    ? unknown extends TInput
      ? (input?: TInput) => Promise<OperationResult<TOutput, TResponse>>
      : (input: TInput) => Promise<OperationResult<TOutput, TResponse>>
    : never;
};

ResourceList ​

列表归一化结果，不伪装 Array。

ts
ResourceList: typeof ResourceList

resourceOf ​

获取实体的无冲突关联入口。

ts
resourceOf: <TModel extends object>(entity: TModel) => EntityReference<TModel>

dataOf ​

获取实体本地数据快照，修改不会自动请求。

ts
dataOf: <TModel>(entity: TModel) => EntityData<TModel>

ResourceReference ​

不可变引用；可推导声明在编译期裁剪能力，动态配置仍需运行时校验。

ts
type ResourceReference<
  TModel = unknown,
  TResponse extends boolean = false,
  TConfig extends ResourceConfig = ResourceConfig,
  TPosition extends ReferencePosition = 'dynamic',
> = ReferenceControls<TModel, TResponse, TConfig, TPosition> &
  Pick<
    ReferenceOperations<TModel, TResponse, TConfig>,
    Extract<DeclaredNames<TConfig>, PositionNames<TPosition>>
  > &
  (TPosition extends 'collection' | 'dynamic'
    ? { (identity: Identity): ResourceReference<TModel, TResponse, TConfig, 'member'> }
    : {}) &
  (string extends keyof Relations<TConfig, TPosition>
    ? {}
    : {
        readonly [TName in keyof Relations<TConfig, TPosition> as TName extends
          | keyof ReferenceControls<TModel, TResponse, TConfig, TPosition>
          | StandardOperation
          | keyof Function
          | keyof Object
          ? never
          : TName]: DefinitionReference<Relations<TConfig, TPosition>[TName], TResponse>;
      });

OperationResult ​

response 模式下的共同结果形态；动态模式返回联合类型。

ts
type OperationResult<TValue, TResponse extends boolean> = TResponse extends true
  ? { data: TValue; response: HttpResponse }
  : TValue;

ResourceEntity ​

普通模型对象的关联类型标记；标量与 unknown 保持原类型。

ts
type ResourceEntity<TModel, TConfig extends ResourceConfig> = TModel extends
  | readonly unknown[]
  | Date
  | RegExp
  | Error
  | ReadonlyMap<unknown, unknown>
  | ReadonlySet<unknown>
  | Blob
  | ArrayBuffer
  ? TModel
  : TModel extends object
    ? TModel &
        EntityRelations<TModel, TConfig> & {
          readonly [entityReferenceType]?: { readonly model: TModel; readonly config: TConfig };
        }
    : TModel;

EntityReference ​

根据实体类型取得成员或单体引用；普通对象保持动态引用兼容。

ts
type EntityReference<TEntity extends object> =
  typeof entityReferenceType extends keyof TEntity
    ? NonNullable<TEntity[typeof entityReferenceType]> extends {
        model: infer TModel;
        config: infer TConfig extends ResourceConfig;
      }
      ? ResourceReference<
          TModel,
          false,
          TConfig,
          TConfig extends { kind: 'singleton' } ? 'singleton' : 'member'
        >
      : ResourceReference<TEntity>
    : ResourceReference<TEntity>;

EntityData ​

去除仅用于类型推导的实体标记。

ts
type EntityData<TEntity> = TEntity extends object
  ? typeof entityReferenceType extends keyof TEntity
    ? NonNullable<TEntity[typeof entityReferenceType]> extends { model: infer TModel }
      ? TModel
      : TEntity
    : TEntity
  : TEntity;

DefinitionReference ​

将声明保留的模型与配置映射为根引用。

ts
type DefinitionReference<TDefinition, TResponse extends boolean = false> =
  TDefinition extends ResourceDefinition<infer TModel, infer TConfig>
    ? ResourceReference<
        TModel,
        TResponse,
        TConfig,
        'kind' extends keyof TConfig
          ? TConfig['kind'] extends 'singleton'
            ? 'singleton'
            : TConfig['kind'] extends 'collection' | undefined
              ? 'collection'
              : 'dynamic'
          : 'collection'
      >
    : ResourceReference;

Identity ​

身份按片段编码，0 有效；复合身份使用有序元组。

ts
type Identity = string | number | readonly (string | number)[];

Query ​

查询对象或显式重复键入口。

ts
type Query = Readonly<Record<string, QueryValue>> | URLSearchParams;

QueryValue ​

query 值，数组代表重复键；undefined 删除，null 编码为空字符串。

ts
type QueryValue =
  | string
  | number
  | boolean
  | null
  | undefined
  | readonly (string | number | boolean | null)[];

executeHttp: (options: HttpOptions, input: HttpRequest, settings?: RequestOptions) => Promise<HttpResponse>

fetchTransport ​

fetchTransport: () => Transport

RequestError ​

统一错误；敏感上下文仅通过显式程序化访问提供，不参与枚举或 JSON。

ts
RequestError: typeof RequestError

RequestErrorKind ​

稳定执行错误分类。

ts
type RequestErrorKind =
  | 'unsupported-operation'
  | 'identity'
  | 'prepare'
  | 'network'
  | 'timeout'
  | 'cancel'
  | 'http'
  | 'business'
  | 'response-shape';

HttpRequest ​

编码完成的请求；Transport 必须监听 signal，且不自动跟随重定向。

ts
interface HttpRequest {
  /** 大写 HTTP 方法。 */
  readonly method: HttpMethod;
  /** 已编码查询的 URL。 */
  readonly url: string;
  /** 小写名称的头快照。 */
  readonly headers: Readonly<Record<string, string>>;
  /** 已编码 body，未提供时不发送。 */
  readonly body?: string | FormData | Blob | ArrayBuffer | undefined;
  /** 请求自身的取消信号；与 RequestOptions.signal 任一取消均生效，不代表服务端撤销。 */
  readonly signal?: AbortSignal | undefined;
}

HttpResponse ​

正常 HTTP 响应，非 2xx 也必须返回此结构。

ts
interface HttpResponse {
  /** HTTP 状态码。 */
  readonly status: number;
  /** 小写头名。 */
  readonly headers: Readonly<Record<string, string>>;
  /** 解码后的响应值。 */
  readonly body?: unknown;
}

Transport ​

只执行最终 HTTP，不读取资源或 scheme；网络失败拒绝 Promise。

ts
type Transport = (request: HttpRequest) => Promise<HttpResponse>;

HttpOptions ​

公共 HTTP 管线设置。

ts
interface HttpOptions {
  /** 唯一请求执行器。 */
  readonly transport: Transport;
  /** 最终快照之前准备请求；每次请求仅调用一次。 */
  readonly prepareRequest?:
    | ((request: HttpRequest) => HttpRequest | Promise<HttpRequest>)
    | undefined;
  /** 只读成功观测，失败被隔离，不重发。 */
  readonly observeResponse?: ((response: HttpResponse) => void | Promise<void>) | undefined;
}

RequestOptions ​

单次执行设置。

ts
interface RequestOptions {
  /** 请求头补丁，undefined 删除已有头。 */
  readonly headers?: Readonly<Record<string, string | undefined>> | undefined;
  /** 调用方拥有的取消信号。 */
  readonly signal?: AbortSignal | undefined;
  /** 等待预算（毫秒），同时向 Transport 发送 abort。 */
  readonly timeout?: number | undefined;
  /** response 模式同时提供适配数据和原始 HTTP 响应；undefined 继承前一层模式。 */
  readonly result?: 'data' | 'response' | undefined;
}
